### SERVIZIO ###########################################################
sudo journalctl -u solar_web.service -f -o cat   # log live
sudo systemctl restart solar_web.service         # restart
sudo systemctl status solar_web.service          # stato + uso memoria

### VENV E LIBRERIE ####################################################
# NB: il solar_web.service versionato dice /home/pi/Desktop/EVR/ ma il deploy
# reale e' altrove. Ricava i percorsi VERI dal service installato:
systemctl cat solar_web.service | grep -E "ExecStart|WorkingDirectory|User"

# Poi imposta queste due variabili di conseguenza:
PROG=~/solar/regolatore-potenza-carica
VENV=$(systemctl cat solar_web.service | grep ExecStart | sed 's/ExecStart=//' | awk '{print $1}')

# Verifica quali librerie sono installate e con che versione
$VENV -c "import telegram, flask, requests, matplotlib, dotenv, sys; \
print('python  ', sys.version.split()[0]); \
print('telegram', telegram.__version__); \
print('flask   ', flask.__version__); \
print('requests', requests.__version__); \
print('matplot ', matplotlib.__version__)"

# Installazione / aggiornamento delle librerie necessarie
$(dirname $VENV)/pip install --upgrade \
    python-telegram-bot flask requests python-dotenv matplotlib

# Verifica sintassi PRIMA di riavviare il servizio
$VENV -m py_compile $PROG/solar_webinterface.py

### DIAGNOSI BOT TELEGRAM ##############################################
# I nomi di comando con MAIUSCOLE sono rifiutati da python-telegram-bot v20+.
# Se in passato il bot non rispondeva, la causa e' questa: cercare l'errore.
sudo journalctl -u solar_web.service --since "-3 days" \
  | grep -iE "not a valid bot command|no current event loop|BOT TELEGRAM ATTIVO|Traceback"

### CENTRALINA EVR 3 W - API RICAVATA (vedi API_CENTRALINA.md) #########
# Endpoint unico: GET http://192.168.1.22/index.json[?btn=<cmd>]
#   i           accendi           o           spegni
#   P<watt>     imposta potenza   L<kWh>      imposta energia da caricare
#   s/e/m/f     algoritmo Sole/Eco/Man/Fast   l  attiva/disattiva limite
# NB: P<watt> funziona SOLO con alg=2 (Man).

curl -s http://192.168.1.22/index.json | python3 -m json.tool
curl -s http://192.168.1.22/info.html | sed 's/<[^>]*>/ /g'   # seriale e firmware

# Campi utili: pcar (potenza auto MISURATA), phome, psun, pnet, energy,
#              status/desc, pwmin/pwmax, alg, limit, tfase, tplug
# La percentuale di carica della batteria NON esiste: serve ISO 15118,
# questa centralina parla IEC 61851 (vedi API_CENTRALINA.md).

### TEST DELLE API #####################################################
IP=192.168.1.23:5000

curl -s http://$IP/api/data | python3 -m json.tool | head -40
curl -s "http://$IP/api/storico?range=24h" | python3 -c "import json,sys; print(len(json.load(sys.stdin)['punti']), 'punti')"
curl -s http://$IP/api/wallbox_raw | python3 -m json.tool

# Salvataggio impostazioni
curl -s -X POST -H 'Content-Type: application/json' \
     -d '{"prelevabile":1500,"protezione":250}' http://$IP/api/settings

# Casi che PRIMA davano HTTP 500 perdendo tutto -> ora 400 con messaggio
curl -s -i -X POST -H 'Content-Type: application/json' -d '{"prelevabile":null}'  http://$IP/api/settings | head -1
curl -s -i -X POST -H 'Content-Type: application/json' -d '{"prelevabile":"abc"}' http://$IP/api/settings | head -1
curl -s -i -X POST -H 'Content-Type: application/json' -d '{"prelevabile":-5}'    http://$IP/api/settings | head -1

# LA PROVA DELLA PERSISTENZA: i valori devono sopravvivere al riavvio
sudo systemctl restart solar_web.service && sleep 12
curl -s http://$IP/api/data | python3 -c "import json,sys; print(json.load(sys.stdin)['config'])"
cat $PROG/config_utente.json

### FILE DI STATO ######################################################
cd $PROG

tail -5 storico.jsonl
ls -lh storico.jsonl                 # cresce ~350 KB/giorno
cat storico_giornaliero.json | python3 -m json.tool
